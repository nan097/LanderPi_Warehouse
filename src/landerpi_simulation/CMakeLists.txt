cmake_minimum_required(VERSION 3.8)
project(landerpi_simulation)
include(cmake/local_dependencies.cmake)
find_package(ament_cmake REQUIRED)
# Gazebo 11 invokes FindPkgConfig under a different package name.
if(DEFINED FPHSA_NAME_MISMATCHED)
  set(_landerpi_previous_name_mismatch "${FPHSA_NAME_MISMATCHED}")
endif()
set(FPHSA_NAME_MISMATCHED TRUE)
find_package(gazebo REQUIRED)
if(DEFINED _landerpi_previous_name_mismatch)
  set(FPHSA_NAME_MISMATCHED "${_landerpi_previous_name_mismatch}")
  unset(_landerpi_previous_name_mismatch)
else()
  unset(FPHSA_NAME_MISMATCHED)
endif()
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(landerpi_grasp SHARED src/grasp_plugin.cpp)
target_include_directories(landerpi_grasp PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(landerpi_grasp ${GAZEBO_LIBRARIES})
ament_target_dependencies(landerpi_grasp gazebo_ros rclcpp std_srvs std_msgs)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(pluginlib REQUIRED)
add_library(landerpi_holonomic_controller SHARED src/holonomic_controller.cpp)
ament_target_dependencies(landerpi_holonomic_controller nav2_core nav2_costmap_2d tf2_geometry_msgs pluginlib)
pluginlib_export_plugin_description_file(nav2_core config/controller_plugin.xml)
install(TARGETS landerpi_holonomic_controller LIBRARY DESTINATION lib)
install(TARGETS landerpi_grasp LIBRARY DESTINATION lib)

install(DIRECTORY launch urdf config worlds rviz media DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/motion_demo.py scripts/sort_runtime.py scripts/scan_filter.py scripts/sort_demo.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/warehouse_joystick.py scripts/warehouse_runtime.py scripts/warehouse_demo.py scripts/warehouse_layout.py DESTINATION lib/${PROJECT_NAME})
ament_package()
