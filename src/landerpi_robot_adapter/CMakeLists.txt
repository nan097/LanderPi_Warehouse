cmake_minimum_required(VERSION 3.8)
project(landerpi_robot_adapter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(landerpi_task_interfaces REQUIRED)
add_executable(base_adapter_node src/base_adapter_node.cpp)
ament_target_dependencies(base_adapter_node rclcpp landerpi_task_interfaces)
install(TARGETS base_adapter_node DESTINATION lib/${PROJECT_NAME})
add_executable(arm_trajectory_adapter_node src/arm_trajectory_adapter_node.cpp)
ament_target_dependencies(arm_trajectory_adapter_node rclcpp landerpi_task_interfaces)
install(TARGETS arm_trajectory_adapter_node DESTINATION lib/${PROJECT_NAME})
add_executable(gripper_adapter_node src/gripper_adapter_node.cpp)
ament_target_dependencies(gripper_adapter_node rclcpp landerpi_task_interfaces)
install(TARGETS gripper_adapter_node DESTINATION lib/${PROJECT_NAME})
add_executable(robot_state_adapter_node src/robot_state_adapter_node.cpp)
ament_target_dependencies(robot_state_adapter_node rclcpp landerpi_task_interfaces)
install(TARGETS robot_state_adapter_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
