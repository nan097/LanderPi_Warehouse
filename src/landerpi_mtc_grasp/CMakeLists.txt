cmake_minimum_required(VERSION 3.8)
project(landerpi_mtc_grasp)
include(../landerpi_simulation/cmake/local_dependencies.cmake)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rclcpp_action control_msgs moveit_task_constructor_core moveit_task_constructor_msgs moveit_ros_planning_interface moveit_msgs geometry_msgs sensor_msgs std_msgs std_srvs gazebo_msgs shape_msgs)

endforeach()
add_executable(color_grasp src/color_grasp.cpp)
target_compile_features(color_grasp PUBLIC cxx_std_17)
ament_target_dependencies(color_grasp rclcpp rclcpp_action control_msgs moveit_task_constructor_core moveit_task_constructor_msgs moveit_ros_planning_interface moveit_msgs geometry_msgs sensor_msgs std_msgs std_srvs gazebo_msgs shape_msgs)
install(TARGETS color_grasp DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/hardware_servo.py scripts/mtc_joystick.py scripts/register_depth.py scripts/color_vision.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/hardware_geometry.py scripts/depth_registration.py scripts/color_geometry.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch rviz DESTINATION share/${PROJECT_NAME})
ament_package()
