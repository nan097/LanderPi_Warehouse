cmake_minimum_required(VERSION 3.8)
project(landerpi_robot_monitor)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(landerpi_task_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(robot_monitor_node src/robot_monitor_node.cpp)
ament_target_dependencies(robot_monitor_node rclcpp landerpi_task_interfaces std_msgs)
install(TARGETS robot_monitor_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
