cmake_minimum_required(VERSION 3.8)
project(landerpi_manipulation)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(landerpi_task_interfaces REQUIRED)
add_executable(manipulation_server src/manipulation_server.cpp)
ament_target_dependencies(manipulation_server rclcpp landerpi_task_interfaces)
install(TARGETS manipulation_server DESTINATION lib/${PROJECT_NAME})
add_executable(planning_scene_bridge src/planning_scene_bridge.cpp)
ament_target_dependencies(planning_scene_bridge rclcpp landerpi_task_interfaces)
install(TARGETS planning_scene_bridge DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
