cmake_minimum_required(VERSION 3.8)
project(landerpi_fine_alignment)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(landerpi_task_interfaces REQUIRED)
add_executable(fine_alignment_server src/fine_alignment_server.cpp)
ament_target_dependencies(fine_alignment_server rclcpp landerpi_task_interfaces)
install(TARGETS fine_alignment_server DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
